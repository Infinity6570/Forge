def specialist_tool_schemas(
    read_only_tools: bool = True,
    allow_workspace_write: bool = False,
    allow_terminal_execution: bool = False,
) -> list[dict[str, Any]]:
    """Build the allowed tool definitions for a specialist based on supervisor authorization."""
    allowed_names = set()
    if read_only_tools:
        allowed_names.update(SPECIALIST_READ_ONLY_TOOLS)
    if allow_workspace_write:
        allowed_names.update(WORKSPACE_WRITE_TOOL_NAMES)
    if allow_terminal_execution:
        allowed_names.update({"run_command", "run_terminal_command"})

    return [
        json.loads(json.dumps(tool))
        for tool in TOOLS
        if tool.get("function", {}).get("name") in allowed_names
