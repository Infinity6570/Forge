     ```css
     @property --cyber-angle {
       syntax: '<angle>';
       initial-value: 0deg;
       inherits: false;
     }

     @keyframes cyber-rotate-glow {
       0% { --cyber-angle: 0deg; }
       100% { --cyber-angle: 360deg; }
     }

     @keyframes cyber-pulse-shimmer {
       0%, 100% { opacity: 0.75; filter: drop-shadow(0 0 12px rgba(0, 240, 255, 0.45)); }
       50% { opacity: 1; filter: drop-shadow(0 0 24px rgba(168, 85, 247, 0.65)) drop-shadow(0 0 35px rgba(0, 240, 255, 0.5)); }
     }

     .cyber-thinking-aura {
       position: relative;
       border-radius: 8px;
     }

     /* Animated glowing border ring */
     .cyber-thinking-aura::before {
       content: '';
       position: absolute;
       inset: -2px;
       border-radius: 10px;
       padding: 2px;
       background: conic-gradient(
         from var(--cyber-angle),
         #00f0ff 0%,
         #7000ff 25%,
         #ff007f 50%,
         #00f0ff 75%,
         #00ffff 100%
       );
       -webkit-mask: linear-gradient(#fff 0 0) content-box, linear-gradient(#fff 0 0);
       -webkit-mask-composite: xor;
       mask-composite: exclude;
       animation: cyber-rotate-glow 3.5s linear infinite, cyber-pulse-shimmer 2.5s ease-in-out infinite;
       pointer-events: none;
       z-index: 5;

     /* Subtle interior ambient back-glow */
     .cyber-thinking-aura::after {
       content: '';
       position: absolute;
       inset: 0;
       border-radius: 8px;

       pointer-events: none;
       z-index: 1;
       animation: cyber-pulse-shimmer 2.5s ease-in-out infinite;
     }
     ```
